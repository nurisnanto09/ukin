Cloud Spreadsheet Sync

Integrasi Google Apps Script (GAS)

Hubungkan portal ke Google Spreadsheet untuk backup nilai, roster, & LKPD.

🟡 Mode Lokal
Status: Siap dihubungkan ke Google Spreadsheet.
📋 Panduan Cepat Pasang Google Apps Script:
1. Buka spreadsheet baru di sheets.new ↗
2. Buka menu Ekstensi → Apps Script, hapus kode bawaan lalu tempel kode di bawah.
3. Klik tombol biru Deploy → New Deployment → Pilih jenis Web App (Pilih: Execute as: Me, Who has access: Anyone). Salin URL Web App dan tempel pada kolom di atas!
📜Kode Google Apps Script (Code.gs): 
/**
 * GOOGLE APPS SCRIPT - EDUASSESS LMS & SIMULATOR PSEUDOCODE UKIN
 * Otomatisasi Sinkronisasi Spreadsheet Google Apps Script:
 * 1. Roster Siswa & Kelompok (1 - 9)
 * 2. Rekap Nilai Pretest, Posttest & Skor HOTS C4-C6
 * 3. Rancangan Pseudocode LKPD Kelompok
 */

function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) || 'ping';
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  // 1. Tarik Data Roster Siswa
  if (action === 'getRoster') {
    var sheet = ss.getSheetByName('Roster_Siswa') || ss.getSheets()[0];
    var data = sheet.getDataRange().getValues();
    var result = [];
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] !== '') {
        result.push({
          absen: Number(data[i][0]),
          nama: String(data[i][1] || ''),
          kelas: String(data[i][2] || 'Informatika - X E7'),
          nis: String(data[i][3] || ''),
          kelompok: String(data[i][4] || ('Kelompok ' + (Math.floor((i - 1) / 4) + 1)))
        });
      }
    }
    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  }

  // 2. Tarik Rekap Nilai
  if (action === 'getRecords') {
    var sheetNilai = ss.getSheetByName('Rekap_Nilai');
    if (!sheetNilai) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
    var dataNilai = sheetNilai.getDataRange().getValues();
    var records = [];
    for (var j = 1; j < dataNilai.length; j++) {
      if (dataNilai[j][0] !== '') {
        records.push({
          absen: Number(dataNilai[j][0]),
          nama: String(dataNilai[j][1] || ''),
          kelas: String(dataNilai[j][2] || ''),
          pretest: Number(dataNilai[j][3] || 0),
          posttest: Number(dataNilai[j][4] || 0),
          accStatus: String(dataNilai[j][5] || 'PENDING')
        });
      }
    }
    return ContentService.createTextOutput(JSON.stringify(records)).setMimeType(ContentService.MimeType.JSON);
  }

  // Ping test
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    message: 'Endpoint GAS EduAssess Aktif & Terhubung!',
    timestamp: new Date().toISOString()
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var raw = e.postData.contents;
    var payload = JSON.parse(raw);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // Simpan Rekap Nilai Murid
    if (payload.action === 'syncAllRecords' && Array.isArray(payload.records)) {
      var sheetNilai = getOrCreateSheet(ss, 'Rekap_Nilai', [
        'No. Absen', 'Nama Siswa', 'Kelas', 'Pretest', 'Posttest',
        'HOTS C4', 'HOTS C5', 'HOTS C6', 'Status ACC', 'Waktu ACC', 'Waktu Pengerjaan'
      ]);

      payload.records.forEach(function(r) {
        var c4 = (r.pretestBloomCorrect && r.pretestBloomCorrect.C4) || 0;
        var c5 = (r.pretestBloomCorrect && r.pretestBloomCorrect.C5) || 0;
        var c6 = (r.pretestBloomCorrect && r.pretestBloomCorrect.C6) || 0;
        sheetNilai.appendRow([
          r.absen,
          r.nama,
          r.kelas || 'Informatika - X E7',
          r.pretestScore || 0,
          r.posttestScore || 0,
          c4 + '/4',
          c5 + '/2',
          c6 + '/4',
          r.accStatus || 'PENDING',
          r.accTimestamp || '-',
          r.timestamp || new Date().toLocaleTimeString('id-ID')
        ]);
      });

      return ContentService.createTextOutput(JSON.stringify({
        status: 'success',
        message: 'Berhasil mencatat ' + payload.records.length + ' rekap nilai ke Google Sheets!'
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // Simpan Rancangan LKPD Kelompok
    if (payload.action === 'saveLKPD' && payload.submission) {
      var sub = payload.submission;
      var sheetLKPD = getOrCreateSheet(ss, 'Rancangan_LKPD_Kelompok', [
        'Kelompok', 'Anggota', 'Kelas', 'Nama Program', 'Deklarasi Variabel', 'Kode Pseudocode', 'Waktu Kirim'
      ]);

      var codeText = Array.isArray(sub.lines) ? sub.lines.join('\n') : String(sub.lines || '');
      sheetLKPD.appendRow([
        sub.namaKelompok || ('Kelompok ' + sub.groupId),
        sub.memberStr || (Array.isArray(sub.members) ? sub.members.join(', ') : '-'),
        sub.kelas || 'Informatika - X E7',
        sub.progName || 'Tentukan_Peringkat_Kelas',
        sub.progDecl || '',
        codeText,
        sub.submittedAt || new Date().toLocaleTimeString('id-ID')
      ]);

      return ContentService.createTextOutput(JSON.stringify({
        status: 'success',
        message: 'Rancangan ' + (sub.namaKelompok || 'Kelompok') + ' berhasil dicatat!'
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // Simpan/Sinkron Roster Murid (36 Siswa)
    if (payload.action === 'saveRoster' && Array.isArray(payload.roster)) {
      var sheetRoster = getOrCreateSheet(ss, 'Roster_Siswa', [
        'Absen', 'Nama Siswa', 'Kelas', 'NIS', 'Kelompok'
      ]);
      if (sheetRoster.getLastRow() > 1) {
        sheetRoster.getRange(2, 1, sheetRoster.getLastRow() - 1, 5).clearContent();
      }
      var rows = payload.roster.map(function(item) {
        return [
          item.absen,
          item.nama,
          item.kelas || 'Informatika - X E7',
          item.nis || '',
          item.kelompok || ('Kelompok ' + (Math.floor((item.absen - 1) / 4) + 1))
        ];
      });
      if (rows.length > 0) {
        sheetRoster.getRange(2, 1, rows.length, 5).setValues(rows);
      }
      return ContentService.createTextOutput(JSON.stringify({
        status: 'success',
        message: 'Berhasil menyimpan ' + rows.length + ' data siswa ke tab Roster_Siswa!'
      })).setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Action tidak dikenali' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#e2e8f0');
    sheet.setFrozenRows(1);
  }
  return sheet;
}
